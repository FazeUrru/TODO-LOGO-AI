                        'seeds': seeds,

    def _monitor_download(self, handle: lt.torrent_handle):
        """Monitorea el progreso de una descarga"""
        try:
            while self.is_running and handle.is_valid():
                s = handle.status()
                
                if s:
                    # Actualizar estadísticas
                    self.downloads[handle]['progress'] = s.progress * 100
                    self.downloads[handle]['downloaded'] = s.total_done
                    self.downloads[handle]['uploaded'] = s.total_upload
                    
                    # Actualizar estado
                    if s.state == lt.torrent_status.seeding:
                        self.downloads[handle]['status'] = 'Completado (seeding)'
                        self.stats['completed_downloads'] += 1
                        self.stats['active_downloads'] -= 1
                        self._procesar_siguiente_en_cola()
                    elif s.state == lt.torrent_status.downloading:
                        self.downloads[handle]['status'] = 'Descargando...'
                    elif s.state == lt.torrent_status.paused:
                        self.downloads[handle]['status'] = 'Pausado'
                    elif s.state == lt.torrent_status.checking_files:
                        self.downloads[handle]['status'] = 'Verificando archivos...'
                    elif s.state == lt.torrent_status.finished:
                        self.downloads[handle]['status'] = 'Completado'
                        self.stats['completed_downloads'] += 1
                        self.stats['active_downloads'] -= 1
                        self._procesar_siguiente_en_cola()
                    
                    # Actualizar lista de archivos
                    if not self.downloads[handle]['file_list']:
                        torrent_info = handle.get_torrent_info()
                        self.downloads[handle]['file_list'] = [
                            {
                                'path': os.path.join(self.categorias[self.downloads[handle]['categoria']], f.path),
                                'size': f.size,
                                'priority': f.priority
                            }
                            for f in torrent_info.files()
                        ]
                
                time.sleep(1)
        except Exception as e:
            logging.error(f"Error al monitorear descarga: {e}")
            self.stats['active_downloads'] -= 1
            self._procesar_siguiente_en_cola()
    
    def _procesar_siguiente_en_cola(self):

    def _verificar_integridad(self, handle: lt.torrent_handle):
        try:
            file_list = handle.get_torrent_info().files()
            save_path = self.downloads[handle]['categoria']
            
            for i in range(file_list.num_files()):
                file_path = os.path.join(save_path, file_list.file_path(i))
                if os.path.exists(file_path):
                    file_hash = hashlib.sha256()
                    with open(file_path, 'rb') as f:
                        while chunk := f.read(8192):
                            file_hash.update(chunk)
                    
                    print(f"{Fore.GREEN}✓ Verificado: {file_list.file_path(i)}{Style.RESET_ALL}")
                    
        except Exception as e:
            logging.error(f"Error en verificación: {e}")
            print(f"{Fore.RED}✗ Error en verificación: {e}{Style.RESET_ALL}")

    def seleccionar_categoria(self, nombre_archivo: str) -> str:
        """Determina la categoría del archivo basado en su nombre"""
        nombre_lower = nombre_archivo.lower()
        if any(palabra in nombre_lower for palabra in ['movie', 'film', 'pelicula', 'cinema']):
            return 'Peliculas'
        elif any(palabra in nombre_lower for palabra in ['serie', 'tv', 'season', 'episode']):
            return 'Series'
        elif any(palabra in nombre_lower for palabra in ['game', 'juego', 'pc game', 'crack']):
            return 'Juegos'
        elif any(palabra in nombre_lower for palabra in ['music', 'mp3', 'album', 'cancion']):
            return 'Musica'
        elif any(palabra in nombre_lower for palabra in ['software', 'program', 'app', 'crack']):
            return 'Programas'
        else:
            return 'Otros'

                        logging.info(f"Descarga completada: {handle.name()}")
                        break
                
                time.sleep(1)
        except Exception as e:
            logging.error(f"Error en monitoreo de descarga: {e}")
            self.downloads[handle]['status'] = f"Error: {str(e)}"
            self.stats['active_downloads'] -= 1
            self._procesar_siguiente_en_cola()

    def _procesar_siguiente_en_cola(self):
        """Procesa el siguiente torrent en la cola si hay espacio"""
        if self.download_queue and self.stats['active_downloads'] < self.max_active_downloads:
            next_magnet = self.download_queue.pop(0)
            self.descargar_torrent(next_magnet)

    def verificar_archivo(self, file_path: str) -> bool:
        """Verifica la integridad de un archivo usando hash SHA256"""
        try:
            sha256_hash = hashlib.sha256()
            with open(file_path, 'rb') as f:
                for byte_block in iter(lambda: f.read(4096), b""):
                    sha256_hash.update(byte_block)
            return True
        except Exception as e:
            logging.error(f"Error al verificar archivo: {e}")
            print(f"{Fore.RED}Error al verificar archivo: {e}{Style.RESET_ALL}")
            return False

    def descargar_torrent(self, magnet_link: str, categoria: str = 'Otros') -> bool:
        if self.stats['active_downloads'] >= self.max_active_downloads:
            print(f"{Fore.YELLOW}Límite de descargas activas alcanzado. Añadiendo a la cola.{Style.RESET_ALL}")
            self.download_queue.append(magnet_link)
            return False
            
        save_path = self.categorias.get(categoria, self.categorias['Otros'])
        
        try:
            info = lt.torrent_info(magnet_link)
            params = {
                'save_path': save_path,
                'ti': info,
                'paused': False,
                'auto_managed': True,
                'duplicate_is_error': True
            }
            
            handle = self.session.add_torrent(params)
            self.downloads[handle] = {
                'name': handle.name(),
                'categoria': categoria,
                'progress': 0,
                'status': 'Descargando...',
                'start_time': datetime.now(),
                'downloaded': 0,
                'uploaded': 0,
                'handle': handle
            }
            self.stats['active_downloads'] += 1
            
            threading.Thread(target=self._monitor_download, args=(handle,), daemon=True).start()
            
            print(f"\n{Fore.GREEN}✓ Descarga iniciada: {handle.name()}{Style.RESET_ALL}")
            time.sleep(1)
            return True
            
        except Exception as e:
            logging.error(f"Error al iniciar descarga: {e}")
            print(f"{Fore.RED}✗ Error al iniciar descarga: {e}{Style.RESET_ALL}")
            return False
    
    def _monitor_download(self, handle: lt.torrent_handle):
        while self.is_running and handle.is_valid():
            s = handle.status()
            if s:
                self.downloads[handle]['progress'] = s.progress * 100
                self.downloads[handle]['downloaded'] = s.total_done
                self.downloads[handle]['uploaded'] = s.total_upload
                
                if s.state == lt.torrent_status.seeding:
                    self.downloads[handle]['status'] = 'Completado (seeding)'
                    self.stats['completed_downloads'] += 1
                    self.stats['active_downloads'] -= 1
                    self._procesar_siguiente_en_cola()
                elif s.state == lt.torrent_status.downloading:
                    self.downloads[handle]['status'] = 'Descargando...'
                elif s.state == lt.torrent_status.paused:
                    self.downloads[handle]['status'] = 'Pausado'
                elif s.state == lt.torrent_status.checking_files:
                    self.downloads[handle]['status'] = 'Verificando archivos...'
                elif s.state == lt.torrent_status.finished:
                    self.downloads[handle]['status'] = 'Completado'
                    self.stats['completed_downloads'] += 1
                    self.stats['active_downloads'] -= 1
                    self._procesar_siguiente_en_cola()
            
            time.sleep(1)
